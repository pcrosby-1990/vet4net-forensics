⏳ Collapse Tone Ritual Scheduler

function scheduleRitual() { const tones = document.getElementById('scheduledTones').value.split(','); const delay = parseInt(document.getElementById('delaySeconds').value, 10) * 1000; const status = document.getElementById('ritualStatus'); status.innerText = `⏳ Ritual scheduled in ${delay / 1000} seconds...`; setTimeout(() => { tones.forEach((tone, i) => { setTimeout(() => { const audio = new Audio(`tones/${tone.trim()}.wav`); audio.play(); }, i * 1000); }); status.innerText = `🔊 Ritual triggered: ${tones.join(", ")}`; }, delay); }
