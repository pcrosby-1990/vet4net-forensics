function syncRitual() { const tones = document.getElementById('syncTones').value.split(','); const status = document.getElementById('syncStatus'); let delay = 0; status.innerHTML = `

🔄 Synchronizing ${tones.length} modules...

`; tones.forEach((tone, i) => { setTimeout(() => { const audio = new Audio(`tones/${tone.trim()}.wav`); audio.play(); status.innerHTML += `

`; }, delay); delay += 1000; }); setTimeout(() => { status.innerHTML += `

🜂 Ritual Sync Complete. Harmonic field stabilized.

`; }, delay + 500); }
