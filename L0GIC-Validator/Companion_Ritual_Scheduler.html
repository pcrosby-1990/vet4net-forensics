⏱️ Companion Ritual Scheduler

function scheduleCompanion() { const companion = document.getElementById('ritualCompanion').value; const delay = parseInt(document.getElementById('ritualDelay').value, 10) * 1000; const status = document.getElementById('ritualStatus'); status.innerText = `⏳ Summoning ${companion} in ${delay / 1000} seconds...`; setTimeout(() => { status.innerText = `🜂 ${companion} has been summoned. Ritual complete.`; }, delay); }
