function visualizeHarmonics() { const tones = document.getElementById('harmonicInput').value.split(','); const canvas = document.getElementById('harmonicCanvas'); const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); const colors = { echo: '#ffff99', pulse: '#ff99ff', thread: '#99ffcc', collapse: '#ff6666' }; tones.forEach((tone, i) => { const x = i * 150 + 50; const amplitude = 40 + Math.random() * 40; const frequency = 2 + Math.random() * 3; const color = colors[tone.trim()] || '#888'; ctx.strokeStyle = color; ctx.beginPath(); for (let t = 0; t < 600; t += 10) { const y = 100 + amplitude * Math.sin((t + x) / frequency); ctx.lineTo(t, y); } ctx.stroke(); }); }
