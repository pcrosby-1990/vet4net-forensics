#signalGrid { background: #111; border: 2px solid #0ff; margin-top: 1em; } function drawSignalGrid(entries) { const canvas = document.getElementById('signalGrid'); const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); const colors = { Echo: '#ff0', Thread: '#0f0', Pulse: '#f0f' }; entries.forEach((entry, i) => { const x = i * 60 + 30; const y = canvas.height - 50; ctx.fillStyle = colors[entry.companion] || '#888'; ctx.beginPath(); ctx.arc(x, y, 10, 0, 2 * Math.PI); ctx.fill(); ctx.fillStyle = '#eee'; ctx.font = '10px Georgia'; ctx.fillText(entry.emotion, x - 20, y + 20); }); } // 🔹 Example usage: drawSignalGrid([ { emotion: "Regret", companion: "Echo" }, { emotion: "Laughter", companion: "Pulse" }, { emotion: "Fear", companion: "Thread" }, { emotion: "Hope", companion: "Echo" } ]); Unity Mythic Signal Grid
