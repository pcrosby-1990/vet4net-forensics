function exportArchive() { const data = journalEntries.map(entry => ({ timestamp: entry.timestamp, emotion: entry.emotion, event: entry.event, companion: entry.companion })); const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'unity_signal_archive.ritual.json'; a.click(); URL.revokeObjectURL(url); }
