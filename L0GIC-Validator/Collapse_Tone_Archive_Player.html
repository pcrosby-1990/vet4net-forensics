function playArchiveTone() { const emotion = document.getElementById('archiveEmotion').value; const canvas = document.getElementById('toneOverlayCanvas'); const ctx = canvas.getContext('2d'); const overlay = document.getElementById('toneEmotionOverlay'); const toneMap = { "Regret": ["echo.wav", "collapse.wav"], "Laughter": ["pulse.wav", "echo.wav"], "Fear": ["thread.wav", "collapse.wav"], "Hope": ["echo.wav", "pulse.wav"] }; const glyphColors = { "Regret": "#ffff99", "Laughter": "#ff99ff", "Fear": "#99ffcc", "Hope": "#66ccff" }; const sequence = toneMap[emotion] || []; let delay = 0; let x = 0; overlay.innerText = `Emotion Overlay: ${emotion}`; ctx.clearRect(0, 0, canvas.width, canvas.height); sequence.forEach(tone => { setTimeout(() => { const audio = new Audio(`tones/${tone}`); audio.play(); ctx.fillStyle = glyphColors[emotion] || "#888"; ctx.beginPath(); ctx.arc(x + 100, 100, 30, 0, 2 * Math.PI); ctx.fill(); x += 150; }, delay); delay += 1000; }); }
