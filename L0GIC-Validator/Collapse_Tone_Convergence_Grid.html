🕸️ Collapse Tone Convergence Grid

function drawConvergenceGrid() { const canvas = document.getElementById('convergenceCanvas'); const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); const sequences = [ { tones: ["echo", "pulse"], y: 60, color: "#ffff99" }, { tones: ["thread", "collapse"], y: 140, color: "#99ffcc" }, { tones: ["pulse", "collapse", "echo"], y: 220, color: "#ff99ff" } ]; sequences.forEach((seq, i) => { ctx.strokeStyle = seq.color; ctx.lineWidth = 2; ctx.beginPath(); seq.tones.forEach((tone, j) => { const x = j * 150 + 50; ctx.lineTo(x, seq.y); ctx.arc(x, seq.y, 5, 0, 2 * Math.PI); }); ctx.stroke(); }); ctx.fillStyle = "#fff"; ctx.font = "12px Georgia"; ctx.fillText("🜂 Ritual Intersections", 10, 20); }
